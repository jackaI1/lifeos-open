cmake_minimum_required(VERSION 3.16)
project(raytrace_trace_analyzer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ZLIB REQUIRED)
add_executable(raytrace src/main.cpp)
target_link_libraries(raytrace PRIVATE ZLIB::ZLIB)
add_executable(geometry_test tests/geometry_test.cpp)
target_include_directories(geometry_test PRIVATE src)
foreach(target raytrace geometry_test)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(ENABLE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
if(ENABLE_SANITIZERS)
  foreach(target raytrace geometry_test)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME geometry COMMAND geometry_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME python_pipeline COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
set_tests_properties(python_pipeline PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
  ENVIRONMENT "RAYTRACE_BINARY=$<TARGET_FILE:raytrace>;MPLBACKEND=Agg")
